# Dorka zsh shell-ready wrapper
__dorka_usable_zdotdir() {
  [[ -n "${1:-}" ]] || return 1
  # Dorka's own dir, by marker file or by the shape older builds wrote.
  [[ "$1" != */shell-ready/zsh ]] || return 1
  [[ ! -f "$1/.dorka-shell-wrapper" ]] || return 1
  # A directory holding no zsh startup file at all is not a config root,
  # whoever wrote it — and a stale value pointing at one would stop zsh from
  # ever reading the user's real .zshenv.
  local _dorka_startup_file
  for _dorka_startup_file in .zshenv .zshrc .zprofile .zlogin; do
    [[ -r "$1/$_dorka_startup_file" ]] && return 0
  done
  return 1
}
if __dorka_usable_zdotdir "${DORKA_ORIG_ZDOTDIR:-}"; then
  builtin export ZDOTDIR="$DORKA_ORIG_ZDOTDIR"
else
  builtin unset ZDOTDIR
fi
builtin unset DORKA_ORIG_ZDOTDIR DORKA_ZSHENV_SOURCE_DIR
builtin unfunction __dorka_usable_zdotdir
builtin typeset -ga _dorka_shell_features
_dorka_shell_features=(${(s:,:)${DORKA_SHELL_FEATURES:-}})
builtin unset DORKA_SHELL_FEATURES
# Why DORKA_HISTFILE is consumed HERE and not in the deferred hook: a user config
# that replaces precmd_functions wholesale drops the hook, and an exported value
# nothing will ever consume is then inherited by every child of this pane,
# including a nested Dorka (#11146). Captured non-exported, it cannot escape.
builtin typeset -g _dorka_histfile="${DORKA_HISTFILE:-}"
builtin unset DORKA_HISTFILE
__dorka_has_feature() { (( ${_dorka_shell_features[(Ie)$1]} )) }
__dorka_has_feature identity && printf "\033]777;dorka-shell-start:%s\007" "$$"
__dorka_osc133_precmd() {
  local exit_code=$?
  if [[ -n "${__dorka_in_command:-}" ]]; then
    builtin printf "\033]133;D;%s\007" "$exit_code"
    builtin unset __dorka_in_command
  fi
  builtin printf "\033]133;A\007"
}
__dorka_osc133_preexec() {
  builtin printf "\033]133;C\007"
  # Why typeset -g: a plain assignment here creates a global inside a function,
  # which prints a warning above every command under warn_create_global.
  builtin typeset -g __dorka_in_command=1
}
__dorka_deferred_init() {
  # Why first: this body runs after the user's own config, so it would otherwise
  # inherit whatever options that config left set. Under NO_UNSET an unset
  # precmd_functions is fatal, and KSH_ARRAYS makes the 1-based feature lookup
  # drop whichever feature is listed first.
  builtin emulate -L zsh
  (( $+_dorka_deferred_init_done )) && return 0
  builtin typeset -g _dorka_deferred_init_done=1
  builtin typeset -g precmd_functions
  if __dorka_has_feature markers; then
    precmd_functions=(${precmd_functions:/__dorka_deferred_init/__dorka_osc133_precmd})
    preexec_functions=(__dorka_osc133_preexec ${preexec_functions[@]})
  else
    precmd_functions=(${precmd_functions:#__dorka_deferred_init})
  fi
  if __dorka_has_feature overlay; then
    # Why: ~/.zshrc can export the user's default OpenCode config after spawn.
    __dorka_restore_agent_teams_path() {
      [[ -n "${DORKA_AGENT_TEAMS_SHIM_DIR:-}" ]] || return 0
      case "$PATH" in
        "${DORKA_AGENT_TEAMS_SHIM_DIR}"|"${DORKA_AGENT_TEAMS_SHIM_DIR}:"*) return 0 ;;
      esac
      export PATH="${DORKA_AGENT_TEAMS_SHIM_DIR}:$PATH"
    }
    __dorka_restore_agent_teams_path
    [[ -n "${DORKA_OPENCODE_CONFIG_DIR:-}" ]] && export OPENCODE_CONFIG_DIR="${DORKA_OPENCODE_CONFIG_DIR}"
    [[ -n "${DORKA_MIMOCODE_HOME:-}" ]] && export MIMOCODE_HOME="${DORKA_MIMOCODE_HOME}"
    # Why: OMP does not auto-load Dorka's managed status extension; wrap only
    # interactive launch invocations so subcommands such as `omp config` keep
    # their normal argv shape.
    __dorka_omp_should_skip_extension() {
      case "${1:-}" in
        'help'|'--help'|'-h'|'--version'|'-v') return 0 ;;
        '__complete'|'acp'|'agents'|'auth-broker'|'auth-gateway'|'bench'|'commit'|'completions'|'config'|'dry-balance'|'gallery'|'grep'|'grievances'|'install'|'join'|'models'|'plugin'|'read'|'say'|'search'|'setup'|'shell'|'ssh'|'stats'|'tiny-models'|'token'|'ttsr'|'update'|'usage'|'worktree'|'q'|'wt') return 0 ;;
      esac
      return 1
    }
    __dorka_omp_cwd_is_usable() {
      local __dorka_physical_cwd
      [[ -x . ]] || return 1
      if [[ -n "${PWD:-}" && -d "${PWD:-}" ]]; then
        [[ "${PWD}" -ef . ]]
      else
        # Why compare the path: shell builtins can print a cached path for a deleted cwd.
        __dorka_physical_cwd="$(builtin pwd -P 2>/dev/null)" || return 1
        [[ -d "$__dorka_physical_cwd" && "$__dorka_physical_cwd" -ef . ]]
      fi
    }
    __dorka_omp_invoke() {
      local __dorka_use_extension="$1"
      shift
      if [[ $__dorka_use_extension -eq 1 && -n "${DORKA_OMP_STATUS_EXTENSION:-}" && -f "${DORKA_OMP_STATUS_EXTENSION}" ]]; then
        if [[ "${1:-}" == "launch" ]]; then
          shift
          command omp launch --extension "${DORKA_OMP_STATUS_EXTENSION}" "$@"
        else
          command omp --extension "${DORKA_OMP_STATUS_EXTENSION}" "$@"
        fi
      else
        command omp "$@"
      fi
    }
    __dorka_omp() {
      local __dorka_use_extension=1
      __dorka_omp_should_skip_extension "${1:-}" && __dorka_use_extension=0
      if ! __dorka_omp_cwd_is_usable; then
        local __dorka_logical_cwd="${PWD:-${DORKA_WORKTREE_PATH:-${DORKA_ROOT_PATH:-}}}"
        # Why: a restored shell can retain the deleted directory inode after its path is recreated.
        (
          if [[ -z "$__dorka_logical_cwd" ]]; then
            printf 'Dorka: OMP cannot start because no terminal working directory is available. Open a new terminal in an existing directory.\n' >&2
            return 1
          fi
          if ! builtin cd -P -- "$__dorka_logical_cwd" 2>/dev/null; then
            printf 'Dorka: OMP cannot access the terminal working directory "%s". Open a new terminal in an existing directory.\n' "$__dorka_logical_cwd" >&2
            return 1
          fi
          __dorka_omp_invoke "$__dorka_use_extension" "$@"
        )
      else
        __dorka_omp_invoke "$__dorka_use_extension" "$@"
      fi
    }
    if [[ -n "${DORKA_OMP_STATUS_EXTENSION:-}" ]]; then
      # Why the function reserved word: it suppresses alias expansion of the name, which
      # an `alias omp` otherwise rewrites at parse time, aborting the rest of the file.
      function omp { __dorka_omp "$@"; }
    fi

    # Why: Codex must keep using Dorka's runtime CODEX_HOME after rc files.
    [[ -n "${DORKA_CODEX_HOME:-}" ]] && export CODEX_HOME="${DORKA_CODEX_HOME}"
    # Why: a typed alias expands inside the shell, after pane launch prep.
    # Why unalias inside the substitution: an alias named codex makes command -v
    # report the alias text, and the subshell leaves the user's own alias intact.
    # Why || : twice — zsh alone aborts inside the substitution, but every shell's
    # assignment adopts its exit status, so an absent codex trips set -e in bash too.
    __dorka_codex_binary="$(unalias codex 2>/dev/null || :; command -v codex 2>/dev/null || :)"
    if [[ -n "${DORKA_CODEX_LAUNCH_PREFLIGHT:-}" && -x "${DORKA_CODEX_LAUNCH_PREFLIGHT}" && -n "${__dorka_codex_binary:-}" && -x "${__dorka_codex_binary}" ]]; then
      # Why the function reserved word: it suppresses alias expansion of the name,
      # which otherwise rewrites this header at parse time and aborts the whole file.
      function codex {
        "${DORKA_CODEX_LAUNCH_PREFLIGHT}" agent hooks prepare-codex >/dev/null 2>&1 || :
        command codex "$@"
      }
    fi
    unset __dorka_codex_binary
  fi
  if [[ -n "${_dorka_histfile:-}" ]]; then
    HISTFILE="$_dorka_histfile"
  fi
  if __dorka_has_feature ready || __dorka_has_feature startup; then
    __dorka_emit_ready_marker=""
    __dorka_has_feature ready && __dorka_emit_ready_marker=1
    # Why: capture the prior zle-line-init so the marker chains to it. On a
    # re-source we are already the bound widget, so keep the function captured
    # the first time instead of clobbering it to empty (which would silently
    # drop the user's widget on every prompt after the second source). Only
    # user-defined widgets are chainable as plain functions; builtin/completion
    # forms (rare for zle-line-init) are left unchained.
    if [[ "${widgets[zle-line-init]:-}" == "user:__dorka_prompt_mark" ]]; then
      :
    elif (( ${+widgets[zle-line-init]} )) && [[ "${widgets[zle-line-init]}" == user:* ]]; then
      __dorka_prev_line_init_fn="${widgets[zle-line-init]#user:}"
    else
      __dorka_prev_line_init_fn=""
    fi
    __dorka_prompt_mark() {
      if [[ "${__dorka_emit_ready_marker-1}" == 1 ]]; then
        printf "\033]777;dorka-shell-ready\007"
      fi
      # Why: call the prior hook as a plain function, not an aliased widget, so
      # $WIDGET stays zle-line-init for add-zle-hook-widget dispatchers.
      if [[ -n "${__dorka_prev_line_init_fn:-}" ]]; then
        "${__dorka_prev_line_init_fn}" "$@"
      fi
      if (( ${+DORKA_POSIX_SHELL_STARTUP_COMMAND} )); then
        BUFFER="$DORKA_POSIX_SHELL_STARTUP_COMMAND"
        CURSOR=${#BUFFER}
        builtin unset DORKA_POSIX_SHELL_STARTUP_COMMAND
        zle accept-line
      fi
    }
    zle -N zle-line-init __dorka_prompt_mark
  fi
  # Why called here: we were appended during this prompt's own precmd sweep, so
  # the permanent hook has not run yet and the first prompt would lose its mark.
  __dorka_has_feature markers && __dorka_osc133_precmd
  builtin unset _dorka_shell_features _dorka_histfile
  builtin unfunction __dorka_deferred_init __dorka_has_feature
}
{
  builtin typeset _dorka_user_zshenv="${ZDOTDIR-$HOME}/.zshenv"
  [[ ! -r "$_dorka_user_zshenv" ]] || builtin source -- "$_dorka_user_zshenv"
} always {
  builtin unset _dorka_user_zshenv
  builtin typeset -ag precmd_functions
  (( ${precmd_functions[(Ie)__dorka_deferred_init]} )) || precmd_functions+=(__dorka_deferred_init)
}
